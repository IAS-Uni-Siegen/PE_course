\task {Inverting buck-boost converter}
An inverting buck-boost converter (see \autoref{fig:Inverting buck-boost converter}) is used to generate the negative supply voltage of an electronic control unit (ECU). The input voltage is specified as $U_\mathrm{1}=\SI{18}{\volt}$, the output voltage is regulated to $U_\mathrm{2}=\SI{12}{\volt}$. The input voltage is considered constant and the output voltage ripple is comparably small not affecting the inductor current ripple. The output power can vary in the range $P_\mathrm{2} = \SI{2}{\watt} \, \dots \, \SI{15}{\watt}$. Initially all components are considered ideal and lossless.

\input{./fig/ex03/Fig_inverseConverter.tex}

\subtask{The system should operate at boundary conduction mode (BCM)   throughout the entire output power range.  Determine the maximum permissible inductance $L$ such that the switching frequency satisfies $f_\mathrm{s}\geq\SI{20}{\kilo\hertz}$.}

\begin{solutionblock}
	With voltages $U_1$ and $U_2$ given in the task, the duty cycle is calculated as:
	\begin{equation}
		D = \frac{U_\mathrm{2}}{U_\mathrm{1}+U_\mathrm{2}} = \frac{\SI{12}{\volt}}{\SI{18}{\volt}+\SI{12}{\volt}} = 0.4.
	\end{equation}
	The (highest) average output current is
	\begin{equation}
		I_\mathrm{2} = \frac{P_\mathrm{2}}{U_\mathrm{2}} = \frac{\SI{15}{\watt}}{\SI{12}{\volt}} = \SI{1.25}{\ampere}
	\end{equation}
	resulting in an average inductor current of
	\begin{equation}
		\overline{i}_\mathrm{L} = \frac{I_\mathrm{2}}{1-D} = \frac{\SI{1.25}{\ampere}}{0.6} = \SI{2.083}{\ampere}.
	\end{equation}
	Due to the triangular shape of the inductor current reaching zero at the beginning and end of each switching period in BCM, the peak-to-peak ripple current is twice the average inductor current:
	\begin{equation}
		\Delta i_{\mathrm{L}} = 2 \overline{i}_\mathrm{L} = \SI{4.17}{\ampere}.
	\end{equation}
	The inverting buck-boost converter's current ripple is defined as
	\begin{equation}
		\Delta i_{\mathrm{L}} = D T_\mathrm{s} \frac{U_1}{L} = \frac{D U_1}{f_\mathrm{s} L},
		\label{eq:sol_task_3_1_buck_boost_current_ripple}
	\end{equation}
	and therefore, the required inductance is calculated as:
	\begin{equation}
		L = \frac{D U_1}{f_\mathrm{s} \Delta i_{\mathrm{L}}} = \frac{0.4 \cdot \SI{18}{\volt}}{\SI{20}{\kilo\hertz} \cdot \SI{4.17}{\ampere}} = \SI{86.4}{\micro\henry}.
	\end{equation}
	It can be noted that the calculation is conducted for the maximum output power as smaller output powers lead to smaller inductor currents and therefore, smaller current ripples. Hence, the maximum power consideration is the worst case for the inductance selection in BCM.
\end{solutionblock}
\subtask{Considering the previous inductance value choice, what is the range of the switching frequency $f_\mathrm{s}$ for the specified output power range to ensure operation in BCM?}

\begin{solutionblock}
	From \eqref{eq:sol_task_3_1_buck_boost_current_ripple} the resulting switching frequency is calculated as:
	\begin{equation}
		f_\mathrm{s} = \frac{D U_1}{L \Delta i_{\mathrm{L}}}.
		\label{eq:sol_task_3_1_buck_boost_switching_frequency}
	\end{equation}
	From the previous subtask we already know the maximum inductor current ripple $\Delta i_{\mathrm{L}}(P_2=\SI{15}{\watt}) = \SI{4.17}{\ampere}$. The minimum inductor current ripple is calculated analogously:
	\begin{equation}
		\Delta i_{\mathrm{L}}(P_2=\SI{2}{\watt}) = 2 \overline{i}_\mathrm{L}(P_2=\SI{2}{\watt}) = 2 \cdot \frac{I_\mathrm{2}(P_2=\SI{2}{\watt})}{1-D} = 2 \cdot \frac{\SI{0.167}{\ampere}}{0.6} = \SI{0.556}{\ampere}.
	\end{equation}
	With \eqref{eq:sol_task_3_1_buck_boost_switching_frequency} the resulting switching frequencies are:
	\begin{equation}
		f_\mathrm{s}(P_\mathrm{2}=\SI{2}{\watt})=\frac{0.4\cdot\SI{18}{\volt}}{\SI{86.4}{\micro\henry}\cdot \SI{0.556}{\ampere}}=\SI{150}{\kilo \hertz},
	\end{equation}
	\begin{equation}
		f_\mathrm{s}(P_\mathrm{2}=\SI{15}{\watt})=\frac{0.4\cdot\SI{18}{\volt}}{\SI{86.4}{\micro\henry}\cdot \SI{4.17}{\ampere}}=\SI{20}{\kilo \hertz}.
	\end{equation}
\end{solutionblock}

\subtask{What is the peak value $\hat i_1=\max\{i_1(t)\}$ of the transistor current?}
\begin{solutionblock}
	During the switch-on period, the transistor current equals
	the inductor current. In BCM, the peak inductor current
	equals the peak-to-peak inductor current ripple. Hence,
	\begin{equation}
		\hat{i}_1
		= \hat{i}_\mathrm{L}
		= \Delta i_\mathrm{L}(P_2=\SI{15}{\watt})
		= \SI{4.17}{\ampere}.
	\end{equation}
	The maximum output power is considered, since the inductor
	current ripple increases linearly with the output power.
\end{solutionblock}

\subtask{How does the duty cycle $D$ change with the output power? Determine the transistor switch-on time $T_\mathrm{on} = D T_\mathrm{s}$ range for minimum and maximum output power.}
\begin{solutionblock}
	Since the average output voltage should remain constant, the duty cycle is independent of the output power and remains constant at $D = 0.4$. However, the switching frequency changes with the output power, which affects the switch-on time.	The previously obtained maximum and minimum switching frequencies lead to:
	\begin{equation}
		T_\mathrm{s}(P_\mathrm{2}=\SI{2}{\watt}) = \frac{1}{\SI{150}{\kilo \hertz}}= \SI{6.67}{\micro \s},
	\end{equation}
	\begin{equation}
		T_\mathrm{s}(P_\mathrm{2}=\SI{15}{\watt}) = \frac{1}{\SI{20}{\kilo \hertz}}= \SI{50}{\micro \s}.
	\end{equation}
	The transistor switch-on times resul
	\begin{equation}
		T_\mathrm{on}(P_\mathrm{2}=\SI{2}{\watt}) = 0.4 \cdot \SI{6.67}{\micro \s} = \SI{2.67}{\micro \s},
	\end{equation}
	\begin{equation}
		T_\mathrm{on}(P_\mathrm{2}=\SI{15}{\watt}) = 0.4 \cdot \SI{50}{\micro \s}= \SI{20}{\micro \s}.
	\end{equation}
\end{solutionblock}

\subtask{At which operating point does the maximum output voltage ripple $\Delta u_\mathrm{2}$ occur (assumption: the load draws a constant current $i_2(t) = I_2$)?}

\begin{solutionblock}
	Assuming a constant load current $I_2$ and that the output
	voltage ripple does not affect the inductor current ripple,
	the capacitor current is given by
	\begin{equation}
		i_\mathrm{C}(t) = \begin{cases}
			-I_2, & t\in [0,T_\mathrm{on}[,            \\
			i_\mathrm{L}(t)-I_2,
			      & t\in [T_\mathrm{on},T_\mathrm{s}].
		\end{cases}
	\end{equation}
	The corresponding signal trajectories in BCM are shown in
	\autoref{fig:ex03-bcm-capacitor-ripple}.

	\input{./fig/ex03/sFig_BCM_Voltage_Ripple.tex}

	The capacitor voltage decreases linearly during the switch-on
	period and follows a quadratic trajectory during the switch-off
	period, since
	\begin{equation}
		C_2\frac{\mathrm{d}u_\mathrm{C}(t)}{\mathrm{d}t}
		=i_\mathrm{C}(t).
	\end{equation}

	The capacitor is charged from $t=T_\mathrm{on}$ until
	$t=t_\mathrm{max}$, when $i_\mathrm{C}(t_\mathrm{max})=0$.
	In BCM, the inductor current decreases linearly during the
	switch-off period from $\hat{i}_\mathrm{L}$
	to zero, hence
	\begin{equation}
		i_\mathrm{L}(t)=
		\hat{i}_\mathrm{L}\frac{T_\mathrm{s}-t}{T_\mathrm{off}},
		\quad t\in [T_\mathrm{on},T_\mathrm{s}].
	\end{equation}

	As the average diode current equals the load current in
	steady state,
	\begin{equation}
		I_2=\overline{i}_\mathrm{D}
		=\frac{1-D}{2}\hat{i}_\mathrm{L}.
	\end{equation}

	Using $i_\mathrm{L}(t_\mathrm{max})=I_2$ yields
	\begin{equation}
		\begin{split}
			t_\mathrm{max}
			 & = T_\mathrm{s}
			-\frac{I_2}{\hat{i}_\mathrm{L}}T_\mathrm{off} \\
			 & = T_\mathrm{s}
			\left(1-\frac{(1-D)^2}{2}\right),
		\end{split}
	\end{equation}
	which corresponds to $t_\mathrm{max}=0.82T_\mathrm{s}$
	for $D=0.4$. The peak-to-peak voltage ripple follows from the charge
	transferred to the capacitor during the charging interval:
	\begin{equation}
		\begin{split}
			\Delta u_\mathrm{C}
			 & = u_\mathrm{C}(t_\mathrm{max})
			-u_\mathrm{C}(T_\mathrm{on})              \\
			 & = \frac{1}{C_2}
			\int_{T_\mathrm{on}}^{t_\mathrm{max}}
			i_\mathrm{C}(t)\,\mathrm{d}t              \\
			 & = \frac{1}{2C_2}
			\left(\hat{i}_\mathrm{L}-I_2\right)
			\left(t_\mathrm{max}-T_\mathrm{on}\right) \\
			 & = \frac{I_2T_\mathrm{s}}{4C_2}(1+D)^2.
		\end{split}
		\label{eq:ex03-bcm-capacitor-ripple}
	\end{equation}

	Since $D$ is constant, $I_2=P_2/U_2$ and
	$f_\mathrm{s}\propto 1/\Delta i_\mathrm{L}\propto 1/P_2$,
	we obtain
	\begin{equation}
		\Delta u_2=\Delta u_\mathrm{C}\propto P_2^2.
	\end{equation}

	Hence, the maximum output voltage ripple occurs at
	$P_2=\SI{15}{\watt}$, corresponding to the minimum switching
	frequency $f_\mathrm{s}=\SI{20}{\kilo\hertz}$.
\end{solutionblock}

\subtask{How high should the output capacitance be selected to ensure $\Delta u_\mathrm{2} < 0.02 \cdot U_\mathrm{2}?$}

\begin{solutionblock}
	Since the output capacitor $C_2$ is connected in parallel
	to the load, the maximum permissible output voltage ripple is
	given by
	\begin{equation}
		\Delta u_{2,\mathrm{lim}}
		= 0.02 U_2
		= 0.02 \cdot \SI{12}{\volt}
		= \SI{0.24}{\volt}.
	\end{equation}

	Using the voltage ripple expression derived in
	\eqref{eq:ex03-bcm-capacitor-ripple} and considering the
	operating point with maximum output power yields
	\begin{equation}
		\Delta u_{2,\mathrm{max}}
		= \frac{\SI{30.625}{\micro\coulomb}}{C_2}
		< \Delta u_{2,\mathrm{lim}}.
	\end{equation}

	Rearranging for the required output capacitance results in
	\begin{equation}
		C_2 >
		\frac{\SI{30.625}{\micro\coulomb}}{\SI{0.24}{\volt}}
		= \SI{127.6}{\micro\farad}.
	\end{equation}
\end{solutionblock}

\subtask{What is the maximum effective value $i_\mathrm{C,RMS}$ of the output capacitor current?}

\begin{solutionblock}
	The RMS value of the capacitor current is defined as
	\begin{equation}
		I_\mathrm{C} =
		\sqrt{\frac{1}{T_\mathrm{s}}
			\int_0^{T_\mathrm{s}} i_\mathrm{C}^2(t)\,\mathrm{d}t}.
	\end{equation}

	Using the capacitor current waveform derived in Subtask~3.1.6
	and the relation $\overline{i}_\mathrm{L}=I_2/(1-D)$ yields
	\begin{equation}
		i_\mathrm{C}(t) =
		\begin{cases}
			-I_2,
			 & t\in[0,T_\mathrm{on}[,            \\[1ex]
			\displaystyle
			\frac{D}{1-D}I_2
			+\Delta i_\mathrm{L}
			\left(\frac{1}{2}
			-\frac{t-T_\mathrm{on}}{T_\mathrm{off}}\right),
			 & t\in[T_\mathrm{on},T_\mathrm{s}].
		\end{cases}
	\end{equation}

	Splitting the integral into the switch-on and switch-off
	intervals results in
	\begin{equation}
		I_\mathrm{C} =
		\sqrt{
			\frac{1}{T_\mathrm{s}}
			\int_0^{T_\mathrm{on}} I_2^2\,\mathrm{d}t
			+
			\frac{1}{T_\mathrm{s}}
			\int_{T_\mathrm{on}}^{T_\mathrm{s}}
			i_\mathrm{C}^2(t)\,\mathrm{d}t
		}.
	\end{equation}

	During the switch-on interval (transistor conducting),
	the capacitor supplies the constant load current:
	\begin{equation}
		\frac{1}{T_\mathrm{s}}
		\int_0^{T_\mathrm{on}} I_2^2\,\mathrm{d}t
		= DI_2^2.
	\end{equation}

	During the switch-off interval, the capacitor current
	decreases linearly. Evaluating the corresponding integral yields
	\begin{equation}
		\begin{split}
			\frac{1}{T_\mathrm{s}}
			\int_{T_\mathrm{on}}^{T_\mathrm{s}}
			i_\mathrm{C}^2(t)\,\mathrm{d}t
			 & =
			\frac{1}{T_\mathrm{s}}
			\int_0^{T_\mathrm{off}}
			\left[
				\frac{D}{1-D}I_2
				+\Delta i_\mathrm{L}
				\left(\frac{1}{2}
				-\frac{t}{T_\mathrm{off}}\right)
				\right]^2
			\mathrm{d}t \\
			 & =
			(1-D)
			\left[
				\left(\frac{D}{1-D}I_2\right)^2
				+\frac{(\Delta i_\mathrm{L})^2}{12}
				\right].
		\end{split}
	\end{equation}

	Combining both contributions, the RMS value of the capacitor
	current is given by
	\begin{equation}
		I_\mathrm{C}
		=
		\sqrt{
			\frac{D}{1-D}I_2^2
			+\frac{1-D}{12}(\Delta i_\mathrm{L})^2
		}.
	\end{equation}

	In BCM, the inductor current ripple is
	$\Delta i_\mathrm{L}=2I_2/(1-D)$, which leads to
	\begin{equation}
		I_\mathrm{C}
		= I_2\sqrt{\frac{1+3D}{3(1-D)}}.
	\end{equation}

	Since the duty cycle $D$ is constant and
	$I_2=P_2/U_2$, the RMS value of the capacitor current
	increases linearly with the output power. Hence, the maximum
	RMS current occurs at $P_2=\SI{15}{\watt}$:
	\begin{equation}
		I_\mathrm{C,max}
		= \SI{1.25}{\ampere}
		\sqrt{\frac{1+3\cdot 0.4}{3(1-0.4)}}
		= \SI{1.38}{\ampere}.
	\end{equation}
\end{solutionblock}

\subtask{Sketch the curves of the voltage $u_\mathrm{T}(t)$ at the power transistor and the current $i_\mathrm{D}(t)$ in the output diode for $P_\mathrm{2}=\SI{2}{\watt}$. Neglecting the voltage output ripple, what is the maximum blocking voltage of the transistor?}

\begin{solutionblock}
	The input voltage equals the sum of the transistor voltage and the inductor voltage:
	\begin{equation}
		U_1 = u_{\mathrm{T}} + u_{\mathrm{L}},
	\end{equation}
	which is rearranged into:
	\begin{equation}
		u_{\mathrm{T}} = U_1 - u_{\mathrm{L}}.
		\label{eq:uT}
	\end{equation}

	During the off time of the transistor, the inductor voltage is defined as
	\begin{equation}
		u_{\mathrm{L}} = -U_2 - u_{\mathrm{D}},
		\label{eq:U2}
	\end{equation}
	where the diode voltage is assumed to be zero during the conduction phase.
	Hence, the maximum blocking voltage of the transistor is calculated with \eqref{eq:uT} and \eqref{eq:U2}, which results in:
	\begin{equation}
		u_{\mathrm{T}} = U_1 - (-U_2)
		= U_1 + U_2 = \SI{18}{\volt} + \SI{12}{\volt} = \SI{30}{\volt}.
	\end{equation}

	The diode current and the blocking voltage of the transistor is shown in \autoref{fig:inverseCon_voltage_current}.
	\input{fig/ex03/sFigDia_Inverse.tex}
\end{solutionblock}

\task {Boost-buck converter and SEPIC topology}

The energy supply of a plasma coating system is realized by a boost converter followed by a buck converter according \autoref{fig:ex03_boost_buck_converter} (with common capacitance).
The converter is connected to a constant voltage $U_\mathrm{1}$ and provides a variable output voltage $U_\mathrm{2}$. The parameters are displayed in \autoref{table:ParametersOfBoostBuckConverter}.

The output voltage is set to the specified value by adjusting the duty cycles $D_\mathrm{1}$ (of transistor $T_\mathrm{1}$)
and $D_\mathrm{2}$ (of transistor $T_\mathrm{2}$) using a control system. Both transistors operate at the same switching frequency and with the
same duty cycle $D_\mathrm{1}$ = $D_\mathrm{2}$ = $D$.
Switching-induced ripple of the intermediate voltage and inductor currents can be neglected unless otherwise stated
The currents in $L_\mathrm{1}$ and $L_\mathrm{2}$ show a continuous course. All components are considered ideal and lossless. The converter operates in steady state.

\subtask{Calculate the duty cycle $D$ range to achieve the stated output voltage $U_\mathrm{2}$ range.}

\begin{solutionblock}
	% Solution
	For the boost-buck converter the voltage transfer ratio delivers:
	\begin{equation}
		\frac{U_\mathrm{2}}{U_\mathrm{1}} = \frac{D} {1-D}
		\quad \Leftrightarrow \quad
		D = \frac{\frac{U_\mathrm{2}}{U_\mathrm{1}}} {1+{\frac{U_\mathrm{2}}{U_\mathrm{1}}}}.
		\label{eq:DutyCycleEx03}
	\end{equation}
	In case of $U_\mathrm{2}$ = $\SI{285}{\volt}$, the duty cycle results in
	\begin{equation}
		D=\frac{\frac{\SI{285}{\volt}}{\SI{380}{\volt}}} {1+{\frac{\SI{285}{\volt}}{\SI{380}{\volt}}}} = 0.429.
	\end{equation}
	For $U_\mathrm{2}$ = $\SI{450}{\volt}$, the duty cycle results in
	\begin{equation}
		D=\frac{\frac{\SI{450}{\volt}}{\SI{380}{\volt}}} {1+{\frac{\SI{450}{\volt}}{\SI{380}{\volt}}}} = 0.542.
	\end{equation}
	Hence, we have $D \in [0.429, 0.542]$ for the specified output voltage range.
\end{solutionblock}

\subtask{Which average intermediate circuit voltage $U_\mathrm{0}$ results depending on $U_\mathrm{2}$?}

\begin{solutionblock}
	% Solution
	The first part is the boost converter part. Hence, the intermediate circuit voltage $U_\mathrm{0}$ is calculated as
	\begin{equation}
		U_\mathrm{0}=U_\mathrm{1} \frac{1}{1-D}.
		\label{eq:DCLinkEx03}
	\end{equation}
	For the above calculated duty cycles, the intermediate circuit voltage results in:
	\begin{equation}
		U_\mathrm{0}(D=0.429)=\SI{380}{\volt} \frac{1}{1-0.429} = \SI{665}{\volt},
	\end{equation}
	\begin{equation}
		U_\mathrm{0}(D=0.542)=\SI{380}{\volt} \frac{1}{1-0.542} = \SI{830}{\volt}.
		\label{eq:sol_task_3_2_max_U0_voltage}
	\end{equation}
\end{solutionblock}

\subtask{What blocking voltage ratings must the transistors $T_\mathrm{1}$ and $T_\mathrm{2}$ and the diodes $D_1$ and
	$D_2$ have?}

\begin{solutionblock}
	% Solution
	Based on the schematic from \autoref{fig:ex03_boost_buck_converter}, following voltage ratings of the semiconductor components can be derived:
	\begin{itemize}
		\item $D_1$: In case of $T_1$ is active, the potential at the $T_1$'s drain is pulled down. The diode $D_1$ has to block $U_0$.
		\item $T_1$: In case $T_1$ is blocking, the inductor current   $i_\mathrm{L1}(t)$ feeds the DC link via $D_1$. The voltage across the ideal diode in forward direction is \SI{0}{\volt}, so that the voltage $U_0$ must be blocked via $T_1$.
		\item $D_2$: In case of $T_2$ is active, the voltage $U_0$ of capacitor $C$ must be blocked via $D_2$.
		\item $T_2$: In case of $T_2$ is blocking, the current $i_\mathrm{L2}$ flows through $D_2$.
		      The forward voltage of the ideal diode is \SI{0}{\volt}, so that $U_0$ is blocked by $T_2$.
	\end{itemize}
	The minimum blocking voltage of all semiconductor components aligns with $U_\mathrm{0}$, which maximally reaches \SI{830}{\volt} according to \eqref{eq:sol_task_3_2_max_U0_voltage}. In real-world applications, additional voltage reserves of the semiconductors must be considered (often 20-30\% of the maximum steady-state voltage) to cover transient overshoot / ringing effects due to parasitics.
\end{solutionblock}

\subtask{Derive the input and output current ripple $\Delta i_\mathrm{L_1}$ and $\Delta i_\mathrm{L_2}$ depending on the duty cycle.}

\begin{solutionblock}
	% Solution
	The current ripple depends on the duty cycle, switching period and inductor voltage.
	For the boost converter stage:
	\begin{equation}
		\Delta i_\mathrm{L1}=\frac{U_\mathrm{1}D}{f_\mathrm{s}L_1}
		\label{eq:Boostripple}
	\end{equation}
	results while the buck converter stage's current ripple is:
	\begin{equation}
		\Delta i_\mathrm{L2}=\frac{\left(U_\mathrm{0}-U_\mathrm{2}\right)D}{f_\mathrm{s}L_2} = \frac{U_\mathrm{1}D}{f_\mathrm{s}L_1} = \Delta i_\mathrm{L1}.
		\label{eq:Buckripple}
	\end{equation}
	Due to the same duty cycle and the same inductor values, the current ripples of both stages are equal.
\end{solutionblock}

\subtask{To what minimum value can the output power be reduced while still ensuring
	continuous operation across the entire output voltage range
	(i.e., continuous current flow in $L_\mathrm{1}$  and $L_\mathrm{2}$)?}

\begin{solutionblock}
	% Solution
	For the boost stage, the power is the product of the input voltage $U_1$ and the average current through $L_1$. Considering boundary conduction mode (BCM) and a triangular	current waveform yields:
	\begin{equation}
		P_\mathrm{min,boost}=U_\mathrm{1}\overline{i}_\mathrm{L1}=U_\mathrm{1}\frac{\Delta i_\mathrm{L1}}{2}.
		\label{eq:PowerBoostripple}
	\end{equation}
	For the buck stage the power is the product of output voltage
	$U_\mathrm{2}$ and average current through $L_2$ considering BCM and a triangular current shape:
	\begin{equation}
		P_\mathrm{min,buck}=U_\mathrm{2}\overline{i}_\mathrm{L2}=U_\mathrm{2}\frac{\Delta i_\mathrm{L2}}{2}.
		\label{eq:PowerBuckripple}
	\end{equation}
	Entering the relevant values in \eqref{eq:PowerBoostripple} and \eqref{eq:PowerBuckripple} leads to \autoref{table:PowerminInputOutputStage}.
	%Solution table Minimal Power 
	\input{./fig/ex03/sFigTab_PowerminInputOutputStage}
	The minimal power, which ensuring continuous operation across the entire output voltage range
	yields  $\SI{1854}{\watt}$. A lower power would violate the $P_{\mathrm{min,buck}}$ limit for $U_\mathrm{2}=\SI{450}{\volt}$, which represents the relevant worst-case condition.
\end{solutionblock}

\subtask{What blocking voltage ratings must the transistor $T$ and the diode $D$ have?}
\begin{solutionblock}
	% Solution
	For the SEPIC topology the transistor blocking voltage is calculated by the sum of
	$U_\mathrm{c}$ and $U_\mathrm{2}$.
	\begin{equation}
		U_\mathrm{T,block}=U_\mathrm{C}+U_\mathrm{2}.
		\label{eq:task_3_2_sol_blocking_voltage_transistor}
	\end{equation}
	Using the volt-second balance of the inductor $L_1$
	\begin{equation}
		DU_\mathrm{1}+\left(1-D\right)\left(U_\mathrm{1}-U_\mathrm{2}-U_\mathrm{C}\right)=0
	\end{equation}
	the voltage $U_\mathrm{C}$ is obtained by
	\begin{equation}
		U_\mathrm{C}=\frac{U_\mathrm{1}}{1-D}-U_\mathrm{2}.
	\end{equation}
	The substitution of $U_\mathrm{1}$ by $U_\mathrm{2}$ leads to
	\begin{equation}
		U_\mathrm{C}=U_\mathrm{2}\frac{\left(1-D\right)}{D}\frac{1}{\left(1-D\right)}-U_\mathrm{2}
		=\left( \frac{1}{D}-1\right) U_\mathrm{2}.
		\label{eq:ucapacitor}
	\end{equation}
	The substitution of $U_\mathrm{C}$ in \eqref{eq:task_3_2_sol_blocking_voltage_transistor} leads to
	\begin{equation}
		U_\mathrm{T,block}=\left( \frac{1}{D}-1\right) U_\mathrm{2}+U_\mathrm{2}=\frac{U_\mathrm{2}}{D}.
	\end{equation}
	The diode has to block the voltage, when the transistor is active, which is the sum of the capacitor and output volage:
	\begin{equation}
		U_\mathrm{D,block} = U_\mathrm{C}+U_\mathrm{2}.
	\end{equation}
	$U_\mathrm{C}$ corresponds to the input voltage $U_\mathrm{1}$, because the voltage transfer ratio of the SEPIC-topology is calculated by:
	\begin{equation}
		\frac{U_\mathrm{2}}{U_\mathrm{1}}= \frac{D}{1-D} \quad \Leftrightarrow \quad
		U_\mathrm{1}=\frac{1-D}{D}U_\mathrm{2}=\left( \frac{1}{D}-1\right)U_\mathrm{2}=U_\mathrm{C}.
	\end{equation}
	The result shows, that the transistor and the diode needs to block the same voltage ($U_\mathrm{T,block}=U_\mathrm{D,block}=\SI{830}{\volt}$) as the sum of input and output voltage, which is the same (idealized) blocking voltage rating as for the transistors and diodes in a boost-buck converter.
\end{solutionblock}

\subtask{Derive the current ripples $\Delta i_\mathrm{L_1}$ and $\Delta i_\mathrm{L_2}$ depending on the duty cycle.}

\begin{solutionblock}
	% Solution
	The current ripple depends on the duty cycle, switching period and inductor voltage.
	If $T$ is active, the current through $L_1$ increases. This corresponds to the ripple current
	\begin{equation}
		\Delta i_\mathrm{L1}=\frac{U_\mathrm{1}D}{f_\mathrm{s}L_\mathrm{1}}.
	\end{equation}
	For $L_2$, the current also increases while $T$ is conducting.
	In this case, the capacitor voltage $U_\mathrm{C}$ is applied
	to $L_2$ in the direction of its current reference arrow.
	The peak-to-peak current ripple is obtained as
	\begin{equation}
		\Delta i_\mathrm{L2}
		= \frac{U_\mathrm{C}D}{f_\mathrm{s}L_2}
		= \frac{U_1D}{f_\mathrm{s}L_2}.
	\end{equation}
\end{solutionblock}

\subtask{To what minimum value can the output power be reduced while still ensuring continuous operation across the entire output voltage range?
	(i.e., continuous current flow through $D$ during the off-time)?}

\begin{solutionblock}
	During the switch-off interval, the diode current is given by the sum of both inductor currents:
	\begin{equation}
		i_\mathrm{D}(t) = i_\mathrm{L1}(t) + i_\mathrm{L2}(t),
		\quad t \in [T_\mathrm{on},T_\mathrm{s}].
	\end{equation}

	In BCM, the diode current reaches zero at the end of the switch-off interval. A corresponding current waveform plot at the BCM
	is depicted in \autoref{fig:ex03-sepic-bcm}.

	\input{./fig/ex03/sFig_SEPIC_BCM.tex}

	Note that the individual inductor currents do not necessarily reach zero at the same time. Since the diode current equals the output current, its average value over one switching period results in
	\begin{equation}
		\overline{i}_2 = \overline{i}_\mathrm{D}
		= \frac{1-D}{2}
		\left(\Delta i_\mathrm{L1}+\Delta i_\mathrm{L2}\right).
	\end{equation}

	Using the current ripple expressions from the previous subtask and the voltage transfer ratio $U_2/U_1=D/(1-D)$, the output power at the BCM boundary is obtained as
	\begin{equation}
		\begin{split}
			P_{2,\mathrm{BCM}}
			 & = U_2 \overline{i}_2                                \\
			 & = \frac{U_2(1-D)}{2}
			\left(\Delta i_\mathrm{L1}+\Delta i_\mathrm{L2}\right) \\
			 & = \frac{U_1^2D^2}{2f_\mathrm{s}}
			\left(\frac{1}{L_1}+\frac{1}{L_2}\right).
		\end{split}
	\end{equation}

	For identical inductances $L_1=L_2=L$, this simplifies to
	\begin{equation}
		P_{2,\mathrm{BCM}} = \frac{U_1^2D^2}{f_\mathrm{s}L}.
	\end{equation}

	Since the BCM boundary power increases monotonically with the duty cycle, the limiting operating point is reached	at the maximum output voltage $U_2=\SI{450}{\volt}$.
	Consequently, the minimum output power ensuring continuous conduction across the entire output voltage range is
	\begin{equation}
		\begin{split}
			P_{2,\mathrm{min}}
			 & = \frac{U_1^2}{f_\mathrm{s}L}
			\left(\frac{U_{2,\mathrm{max}}}
			      {U_1+U_{2,\mathrm{max}}}\right)^2                \\
			 & = \frac{(\SI{380}{\volt})^2}
			     {\SI{50}{\kilo\hertz}\cdot\SI{0.5}{\milli\henry}}
			\left(\frac{450}{380+450}\right)^2                     \\
			 & = \SI{1697.84}{\watt}.
		\end{split}
	\end{equation}

	Hence, the SEPIC allows continuous conduction down to a lower output power than the boost-buck converter.
\end{solutionblock}

\subtask{Describe the advantage and disadvantage of the SEPIC topology
	compared to the boost-buck converter. Consider the necessary components
	and the quality of the output voltage.}

\begin{solutionblock}
	% Solution
	Advantage of SEPIC-topology compared to boost-buck converter:
	\begin{itemize}
		\item The amount of semiconductor components are reduced (only one transistor and diode are needed).
		\item Lower voltage rating requirement for the capacitor ($U_1$ compared to $U_0$).
	\end{itemize}
	Disadvantage of SEPIC-topology compared to boost-buck converter:
	\begin{itemize}
		\item The output current is pulsating (needs more filtering components).
		\item The peak current flowing through the transistor and diode is higher than in the boost-buck converter (higher thermal stress).
	\end{itemize}
\end{solutionblock}
