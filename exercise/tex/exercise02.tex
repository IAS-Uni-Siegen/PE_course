\subtask{Derive the duty cycle $D$ which leads to the specific output voltage of $U_{\mathrm{2}} = \SI{60}{\volt}$.}

\begin{solutionblock}
	\input{./fig/ex02/Fig_ECDTeTa.tex}
	The equivalent circuit diagram (ECD) is shown in \autoref{fig:switching_states_step-down_converter-Ex02}.
	As there are no losses at the transistor, the voltage $U_{\mathrm{1}}$ is equal to $u_{\mathrm{L}}$ for the switch-on period $T_{\mathrm{on}}$ and is defined as
	\begin{equation}
		u_{\mathrm{L}}(t) = U_{\mathrm{1}} = L \frac{\mathrm{d} i_{\mathrm{L}} }{\mathrm{d} t}, \quad T_{\mathrm{on}} < t \leq T_{\mathrm{s}}.
	\end{equation}
	This results in the component differential equations becoming average value equations. This subtask does not consider the complete period $T_{\mathrm{s}}$, but only the switch-on period $T_{\mathrm{on}}$. Furthermore, the initial state of the current $i_{\mathrm{L}}$ must also be considered.
	That is why we can completely rearrange the equation into:
	\begin{equation}
		i_{\mathrm{L}}(T_{\mathrm{on}}) =  \frac{U_{\mathrm{1}} }{L}T_{\mathrm{on}} +  i_{\mathrm{L}}(0).
	\end{equation}
	Applying Kirchhoff's second law, the voltage equation for  the switch-off time $T_{\mathrm{off}}$ is given with:
	\begin{equation}
		U_{\mathrm{1}} = u_{\mathrm{L}} + U_{\mathrm{2}}.
	\end{equation}
	Now, the differential expression for $u_{\mathrm{L}}$ leads to:
	\begin{equation}
		L \frac{\mathrm{d} i_{\mathrm{L}} }{\mathrm{d} t} = U_{\mathrm{1}} - U_{\mathrm{2}}.
	\end{equation}
	The equation for the current $i_{\mathrm{L}}$ at the switch-off time $T_{\mathrm{off}}$ results from the term describing the discharge of the inductance from $T_{\mathrm{off}}$ together with the initial condition of the current at this time
	\begin{equation}
		i_{\mathrm{L}}(t) = \frac{U_{\mathrm{1}}-U_{\mathrm{2}} }{L} (t-T_{\mathrm{on}})+i_{\mathrm{L}}(T_{\mathrm{on}}).
	\end{equation}
	This results in the equation for the current curve during the switch-on period:
	\begin{equation}
		i_{\mathrm{L}}(t) = \frac{U_{\mathrm{1}}-U_{\mathrm{2}} }{L} (t-T_{\mathrm{on}})+\frac{U_{\mathrm{1}}}{L}T_{\mathrm{on}}+i_{\mathrm{L}}(0), \quad T_{\mathrm{on}} < t \leq T_{\mathrm{s}}.
	\end{equation}
	If one considers the steady state $i_{\mathrm{L}}(t=0)=i_{\mathrm{L}}(t=T_{\mathrm{s}})$, this results in
	\begin{equation}
		i_{\mathrm{L}}(0) = \frac{U_{\mathrm{1}}-U_{\mathrm{2}} }{L} (T_{\mathrm{off}})+\frac{U_{\mathrm{1}}}{L}T_{\mathrm{on}}+i_{\mathrm{L}}(0).
	\end{equation}
	Next $i_{\mathrm{L}}(0)$ cancels on both sides due to subtraction
	\begin{equation}
		0 = \frac{U_{\mathrm{1}}-U_{\mathrm{2}} }{L} (T_{\mathrm{off}})+\frac{U_{\mathrm{1}}}{L}T_{\mathrm{on}}.
	\end{equation}
	Canceling out the inductance leads to
	\begin{equation}
		U_{\mathrm{1}}T_{\mathrm{on}}= (-U_{\mathrm{1}}+U_{\mathrm{2}})T_{\mathrm{off}},
	\end{equation}
	and rearranging to:
	\begin{equation}
		U_{\mathrm{1}}(T_{\mathrm{off}}+T_{\mathrm{on}})=  U_{\mathrm{2}}T_{\mathrm{off}}.
	\end{equation}
	The switch-off period $T_{\mathrm{off}}$ plus the switch-on period $T_{\mathrm{on}}$ is the total period $T_{\mathrm{s}}$. Hence, the following equation can be found for the input-output voltage gain:
	\begin{equation}
		\frac{U_{\mathrm{1}}}{U_{\mathrm{2}}}= \frac{T_{\mathrm{off}}}{T_{\mathrm{s}}}= \frac{T_{\mathrm{s}}}{T_{\mathrm{s}}}-\frac{T_{\mathrm{on}}}{T_{\mathrm{s}}} = 1-D.
	\end{equation}
	And with the values given in the task, the duty cycle is calculated as follows:
	\begin{equation}
		D = 1-\frac{U_{\mathrm{1}}}{U_{\mathrm{2}}} = 1- \frac {\SI{12}{\volt}} {\SI{60}{\volt}} = 0.8.
	\end{equation}
\end{solutionblock}

\subtask{Determine the average input current $\overline i_{\mathrm{L}}$.}
\begin{solutionblock}
	With neglecting the losses $(P_{\mathrm{l}} = \SI{0}{\watt})$ of the boost converter, the average current $\overline i_{\mathrm{L}}$ can be calculated using the power balance
	\begin{equation}
		P_{\mathrm{2}} = U_{\mathrm{2}} I_{\mathrm{2}} = {\SI{60}{\volt}} \cdot {\SI{2}{\ampere}} = {\SI{120}{\watt}}.
	\end{equation}
	Since no losses are considered, the power  $P_{\mathrm{1}}$ is equal to  $P_{\mathrm{2}}$.
	This results in the average current $\overline i_{\mathrm{L}}$:
	\begin{equation}
		\overline i_{\mathrm{L}} = \frac{P_{\mathrm{1}}}{U_{\mathrm{1}}}= \frac{\SI{120}{\watt}}{\SI{12}{\volt}} = \SI{10}{\ampere}.
	\end{equation}
	An alternative solution way is to determine the average current $\overline i_{\mathrm{L}}$ via the duty cycle
	\begin{equation}
		\begin{gathered}
			\frac{\overline i_{\mathrm{L}}}{I_{\mathrm{2}}}=\frac{\overline i_{1}}{I_{\mathrm{2}}}=\frac{1}{1-D} \\
			\Leftrightarrow \quad \overline i_{\mathrm{L}}=\frac{I_{\mathrm{2}}}{1-D}=\frac{\SI{2}{\ampere}}{1-0.8}= \SI{10}{\ampere}.
		\end{gathered}
		\label{eq:task_2_1_2_sol_average_inductor_current}
	\end{equation}
\end{solutionblock}

\subtask{ Define a suitable inductance for the coil $L$, so that the boost converter is operating in boundary conduction mode (BCM) when supplying the minimum output power $P_\mathrm{2,min}$. Determine the maximal switch-off current $i_{\mathrm{T}}$ of the transistor $T$ for the rated output current $I_{\mathrm{2}} = \SI{2}{\ampere}$.}
\begin{solutionblock}
	Because no losses are assumed, $P_{\mathrm{1,min}}=P_{\mathrm{2,min}}$ follows. The average current $\overline i_{\mathrm{1,min}}$ over a period to achieve the minimum input power is
	\begin{equation}
		\overline i_{\mathrm{1,min}} = \frac{P_{\mathrm{1,min}}}{U_{\mathrm{1}}} = \frac{\SI{10}{\watt}}{\SI{12}{\volt}}= \SI{0.833}{\ampere}.
	\end{equation}
	The requirement for BCM is that the current is zero at the starting point of the period  $i_\mathrm{1,BCM}(t=0)=\SI{0}{\ampere}$.
	Due to the triangular current waveform of the inductor current $i_{\mathrm{L}}$, the maximum current is given with:
	\begin{equation}
		i_\mathrm{1,BCM}(T_{\mathrm{on}})= 2 \overline i_{\mathrm{1,min}}= 2\cdot \SI{0.833}{\ampere} = \SI{1.667}{\ampere}.
	\end{equation}
	The switch-on period can be determined by the known duty cycle $D$ and the known frequency $f_{\mathrm{s}}$
	\begin{equation}
		T_{\mathrm{on}} = D T_{\mathrm{s}} = \Delta t = \frac{D}{f_{\mathrm{s}}} = \frac{0.8}{\SI{100}{\kilo\hertz}}=\SI{8}{\micro\second}.
	\end{equation}
	The necessary inductance $L$ can be determined from the ripple current $\Delta i_\mathrm{L}$ and the voltage $u_\mathrm{L}$ over the inductor during the switch-on period
	\begin{equation}
		\Delta i_\mathrm{L} = \frac{U_1}{L}T_{\mathrm{on}}\stackrel{\mathrm{BCM}}{=}2\overline{i}_\mathrm{1} \quad \Leftrightarrow \quad L= \frac{D U_\mathrm{1}}{f_{\mathrm{s}}2\overline{i}_{\mathrm{1,min}}} = \frac{0.8 \cdot \SI{12}{\volt}}{\SI{100}{\kilo\hertz\cdot 2\cdot \SI{0.833}{\ampere}}} = \SI{57.6}{\micro\henry}.
	\end{equation}
	The maximum current flows through the transistor at the end of the switch-on interval, i.e., at the upper peak of the triangular current waveform. For nominal output load, the average input current is $\overline i_\mathrm{1} = \SI{10}{\ampere}$, compare \eqref{eq:task_2_1_2_sol_average_inductor_current}. Hence, in this case the converter is operating in CCM and the maximum current through the transistor is given by the superposition of the average input current and the half of the ripple current
	\begin{equation}
		i_\mathrm{T,max} = i_\mathrm{L,max}=\overline i_\mathrm{1} + \frac{\Delta i_\mathrm{L}}{2} = \SI{10}{\ampere} + \frac{\SI{1.667}{\ampere}}{2} = \SI{10.833}{\ampere}.
	\end{equation}
\end{solutionblock}

\subtask{Calculate a suitable capacitance to meet the output voltage ripple specification. Assume that the previously calculated inductor current ripple and the constant output current are not affected by the output voltage ripple. Determine the current stress of the capacitor, i.e., its RMS current $I_\mathrm{C}$.}
\begin{solutionblock}
	The differential form of the capacitor equation is used for this task
	\begin{equation}
		i_\mathrm{C}=C \frac{\mathrm{d}u_\mathrm{c}(t)}{\mathrm{d}t}.
	\end{equation}
	During on-state of the transistor, the output capacitor feed the load. Assuming a constant output current and an ideal capacitor, the output can be adopted as linearly decreasing for this interval, i.e., the voltage ripple directly follows from the constant load current during the switch-on period:
	\begin{equation}
		\Delta u_\mathrm{C} = \frac{1}{C} \int_{0}^{T_\mathrm{on}} i_\mathrm{C}(t) \mathrm{d}t = \frac{1}{C} \int_{0}^{T_\mathrm{on}} I_\mathrm{2} \mathrm{d}t = \frac{I_\mathrm{2}}{C} T_\mathrm{on} = \frac{I_\mathrm{2}}{C} \frac{D}{f_\mathrm{s}}.
	\end{equation}
	Rearranging the equation, the capacitance can be determined as
	\begin{equation}
		C = \frac{I_\mathrm{2}D}{\Delta u_\mathrm{C}f_\mathrm{s}}=\frac{\SI{2}{\ampere}\cdot 0.8}{\SI{120}{\milli\volt} \cdot \SI{100}{\kilo\hertz}}=\SI{133.333}{\micro\farad}.
	\end{equation}
	An important stress parameter for dimensioning the capacitor is the RMS value of its current:
	\begin{equation}
		I_\mathrm{C}=\sqrt{\frac{1}{T_\mathrm{s}}\int_{0}^{T_\mathrm{s}}i_\mathrm{C}^2(t)\mathrm{d}t}.
	\end{equation}
	Splitting the integral into the switch-on and switch-offs periods, we can rewrite as:
	\begin{equation}
		I_\mathrm{C}=\sqrt{\frac{1}{T_\mathrm{s}}\int_{0}^{DT_\mathrm{s}}i_\mathrm{C}^2(t)\mathrm{d}t+\frac{1}{T_\mathrm{s}}\int_{DT_\mathrm{s}}^{T_\mathrm{s}}i_\mathrm{C}^2(t)\mathrm{d}t}.
	\end{equation}
	For the switch-on period, the capacitor current is equal to the negative output current $i_\mathrm{C}(t)=-I_\mathrm{2}$:
	\begin{equation}
		\frac{1}{T_\mathrm{s}}\int_{0}^{DT_\mathrm{s}}i_\mathrm{C}^2(t)\mathrm{d}t=\frac{1}{T_\mathrm{s}}\int_{0}^{DT_\mathrm{s}}I_\mathrm{2}^2\mathrm{d}t=\frac{1}{T_\mathrm{s}}I_\mathrm{2}^2DT_\mathrm{s}=DI_\mathrm{2}^2.
	\end{equation}
	For the switch-off period, the capacitor current is
	\begin{equation}
		i_\mathrm{C}(t)= i_\mathrm{L}(t) - I_2 = \overline{i}_\mathrm{L} + \frac{\Delta i_\mathrm{L}}{2} -\frac{\Delta i_\mathrm{L}}{T_\mathrm{off}}(t-T_\mathrm{on})- I_2, \quad t \in [DT_\mathrm{s},T_\mathrm{s}]
	\end{equation}
	representing a linear decrease of the inductor current from its maximum value to its minimum value. Inserting $i_\mathrm{C}(t)=i_\mathrm{L}(t) - I_\mathrm{2}$ into the integral gives
	\begin{equation}
		\begin{aligned}
			\frac{1}{T_\mathrm{s}}\int_{DT_\mathrm{s}}^{T_\mathrm{s}}i_\mathrm{C}^2(t)\,\mathrm{d}t
			 & =\frac{1}{T_\mathrm{s}}\int_{0}^{T_\mathrm{off}}
			\left(\overline{i}_\mathrm{L} + \frac{\Delta i_\mathrm{L}}{2} -\frac{\Delta i_\mathrm{L}}{T_\mathrm{off}}t- I_2\right)^2\mathrm{d}t \\
			 & =(1-D)\left[\left(\overline{i}_\mathrm{L}-I_\mathrm{2}\right)^2
				         +\frac{(\Delta i_\mathrm{L})^2}{12}\right].
		\end{aligned}
	\end{equation}
	Therefore,
	\begin{equation}
		\begin{aligned}
			I_\mathrm{C}
			 & =\sqrt{D I_\mathrm{2}^2+(1-D)\left[\left(\overline{i}_\mathrm{L}-I_\mathrm{2}\right)^2
					    +\frac{(\Delta i_\mathrm{L})^2}{12}\right]} \\
			 & =\sqrt{0.8(\SI{2}{\ampere})^2+0.2\left[
					    (\SI{10}{\ampere}-\SI{2}{\ampere})^2+
					    \frac{(\SI{1.667}{\ampere})^2}{12}\right]}
			=\SI{4.006}{\ampere}.
		\end{aligned}
	\end{equation}
\end{solutionblock}

\subtask{First, only the influence of the inductor resistance $R_\mathrm{L}$ is analyzed (i.e. the other internal resistances are neglected). Derive the efficiency $\eta$ and voltage ratio $U_\mathrm{2}/U_\mathrm{1}$ of the boost converter in dependence on the duty cycle $D$ and the resistance ratio $\alpha= R_\mathrm{L}/R$. Sketch both functions, $\eta$ and  $U_\mathrm{2}/U_\mathrm{1}$ over the duty cycle $D$ for $\alpha=\nicefrac{1}{150}$.}
\begin{solutionblock}
	The efficiency is given by
	\begin{equation}
		\eta = \frac{P_\mathrm{2}}{P_\mathrm{1}} = \frac{P_\mathrm{2}}{P_\mathrm{2}+P_\mathrm{l}} = \frac{U_\mathrm{2} I_\mathrm{2}}{U_\mathrm{2} I_\mathrm{2} +R_\mathrm{L} i_\mathrm{L}^2} \label{eq:efficiency}.
	\end{equation}
	with $P_\mathrm{l}$ being the power loss in the inductor. 	The capacitor is charged when it is switched off and discharged when it is switched on. The same applies to the inductor, which discharges when it is switched off. Hence, the average current $I_\mathrm{2}$ can be expressed as
	\begin{equation}
		I_\mathrm{2}=\frac{1}{T_\mathrm{s}} \int_{0}^{DT_\mathrm{s}} i_\mathrm{C, discharge}(t)\mathrm{d}t-\frac{1}{T_\mathrm{s}} \int_{DT_\mathrm{s}}^{T_\mathrm{s}} i_\mathrm{C,charge}(t)\mathrm{d}t+\frac{1}{T_\mathrm{s}} \int_{DT_\mathrm{s}}^{T_\mathrm{s}} I_\mathrm{L}(t)\mathrm{d}t.
		\label{eq:task_2_2_1_output_current_average}
	\end{equation}
	As the moved charges during charging and discharging of the capacitor must be equal (otherwise the average voltage across the capacitor would change between periods, i.e., no steady state), the first two integrals cancel each other out. This leads to
	\begin{equation}
		\frac{1}{T_\mathrm{s}}\int_{0}^{T_s} i_\mathrm{C}(t)\mathrm{d}t \stackrel{!}{=} 0 = \frac{1}{T_\mathrm{s}} \int_{0}^{DT_\mathrm{s}} i_\mathrm{C, discharge}(t)\mathrm{d}t-\frac{1}{T_\mathrm{s}} \int_{DT_\mathrm{s}}^{T_\mathrm{s}} i_\mathrm{C,charge}(t)\mathrm{d}t.
	\end{equation}
	This simplifies the expression \eqref{eq:task_2_2_1_output_current_average} to
	\begin{equation}
		I_\mathrm{2}= \frac{1}{T_\mathrm{s}} \int_{DT_\mathrm{s}}^{T_\mathrm{s}} i_\mathrm{L}(t)\mathrm{d}t.
	\end{equation}
	As the input current (which equals the inductor current) is constant $i_\mathrm{L}(t)=I_1$, the integral solution is straightforward and leads to:
	\begin{equation}
		I_\mathrm{2}= \frac{I_1}{T_\mathrm{s}}(T_\mathrm{s}-(DT_\mathrm{s}))=I_1(1-D) \label{eq:current_1}.
	\end{equation}
	Next, we insert \eqref{eq:current_1} into \eqref{eq:efficiency} and receive:
	\begin{equation}
		\eta=\frac{U_\mathrm{2}I_\mathrm{2}}{U_\mathrm{2}I_\mathrm{2}+\frac{ R_\mathrm{L}I_\mathrm{2}^2}{(1-D)^2}}=\frac{U_\mathrm{2}}{U_\mathrm{2}+\frac{ R_\mathrm{L}I_\mathrm{2}}{(1-D)^2}}=\frac{RI_\mathrm{2}}{RI_\mathrm{2}+\frac{ R_\mathrm{L}I_\mathrm{2}}{(1-D)^2}}=\frac{R}{R+\frac{ R_\mathrm{L}}{(1-D)^2}}=\frac{1}{1+\frac{\alpha}{(1-D)^2}}.
		\label{eq:efficiencyOverD}
	\end{equation}
	From the general definition of efficiency
	\begin{equation}
		\eta=\frac{P_\mathrm{2}}{P_\mathrm{1}} = \frac{U_\mathrm{2}I_\mathrm{2}}{U_\mathrm{1}I_1}
	\end{equation}
	the voltage ratio can be expressed as
	\begin{equation}
		\frac{U_\mathrm{2}}{U_\mathrm{1}}=\eta \frac{I_1}{I_\mathrm{2}}= \eta \frac{1}{(1-D)}=\frac{1}{1+\frac{\alpha}{(1-D)^2}}\frac{1}{1-D}=\frac{1}{(1-D)+\frac{\alpha}{1-D}}\cdot \frac{(1-D)}{(1-D)}=\frac{1-D}{(1-D)^2+\alpha}.
		\label{eq:voltageRatioOverD}
	\end{equation}
	The found results are visualized in \autoref{fig:voltageRatioAndEfficiency}.
	\input{fig/ex02/sFig_boost_voltage_efficiency.tex}
\end{solutionblock}

\subtask{For $R_\mathrm{L}=\SI{0.2}{\ohm}$, determine the required duty cycle value for the given input and output voltages.}
\begin{solutionblock}
	Defining the auxiliary duty cycle $D_\mathrm{P}=1-D$ we can rewrite the voltage gain expression as:
	\begin{equation}
		\frac{U_\mathrm{2}}{U_\mathrm{1}}=\frac{D_\mathrm{P}}{(D_\mathrm{P})^2+\alpha}.
	\end{equation}
	As this expression is a quadratic term, the quadratic formula  can be used to solve for $D_\mathrm{P}$
	\begin{equation}
		\frac{U_\mathrm{2}}{U_\mathrm{1}}(D_\mathrm{P})^2-D_\mathrm{P}+\frac{U_\mathrm{2}\alpha}{U_\mathrm{1}}=0,
	\end{equation}
	\begin{equation}
		D_\mathrm{P}^2 - \frac{U_\mathrm{1}}{U_\mathrm{2}} D_\mathrm{P}+\alpha=0. \label{eq:dutycycle under consideration of inductor losses}
	\end{equation}
	The first term resulting from \eqref{eq:dutycycle under consideration of inductor losses} is:
	\begin{equation}
		D_\mathrm{P1}=\frac{1}{2}\frac{U_\mathrm{1}}{U_\mathrm{2}}+\sqrt{\left(\frac{1}{2}\frac{U_\mathrm{1}}{U_\mathrm{2}}\right)^2-\alpha}= \frac{1}{2}\cdot\frac{\SI{12}{\volt}}{\SI{60}{\volt}}+\sqrt{\left({\frac{1}{2}}\cdot\frac{\SI{12}{\volt}}{\SI{60}{\volt}}\right)^2-\frac{\SI{0.2}{\ohm}}{\SI{30}{\ohm}}}= 0.158.
	\end{equation}
	If one inserts this into the equation $D_\mathrm{1}=1-D_\mathrm{P1}$, one gets the result $D_\mathrm{1}=0.842$.

	The second term resulting from \eqref{eq:dutycycle under consideration of inductor losses} is:
	\begin{equation}
		D_\mathrm{P2}=\frac{1}{2}\frac{U_\mathrm{1}}{U_\mathrm{2}}-\sqrt{\left(\frac{1}{2}\frac{U_\mathrm{1}}{U_\mathrm{2}}\right)^2-\alpha}= \frac{1}{2}\cdot\frac{\SI{12}{\volt}}{\SI{60}{\volt}}-\sqrt{\left({\frac{1}{2}}\cdot\frac{\SI{12}{\volt}}{\SI{60}{\volt}}\right)^2-\frac{\SI{0.2}{\ohm}}{\SI{30}{\ohm}}}= 0.042.
	\end{equation}
	This corresponds to $D_\mathrm{2}=0.958$. Since both duty cycles are valid in terms of achieving the desired output voltage, a loss analysis is necessary to determine the better operating point choice. The inductor currents for the two above considered cases are:
	\begin{equation}
		I_\mathrm{L1} = \frac{I_\mathrm{2}}{1-D_\mathrm{1}}= \frac{\SI{2}{\ampere}}{1-0.842}=\SI{12.679}{\ampere},
	\end{equation}
	\begin{equation}
		I_\mathrm{L2} = \frac{I_\mathrm{2}}{1-D_\mathrm{2}}= \frac{\SI{2}{\ampere}}{1-0.958}=\SI{47.321}{\ampere}.
	\end{equation}
	Putting these current values in relation to the losses, the following can be determined by using the relation $P=I^2R$:
	\begin{equation}
		\frac{(\SI{47.321}{\ampere})^2}{(\SI{12.679}{\ampere})^2}=13.9296.
	\end{equation}
	The losses are 14 times higher by a duty cycle of $D_\mathrm{2}=0.958$ compared to $D_\mathrm{1}=0.842$, i.e., the first duty cycle is clearly the better choice.
\end{solutionblock}

\subtask{Calculate the efficiencies $\eta_1(D_1)$ and $\eta_2(D_2)$ of the boost converter for an output current of $\SI{2}{\ampere}$ and an inductor resistance of $R_{\mathrm{L}} = \SI{0.2}{\Omega}$.}

	\end{equation}

\end{solutionblock}

\subtask{In addition, consider the conduction losses of the diode $D$ and the transistor $T$. Assume an equivalent resistance of $R_{\mathrm{D}} = \SI{0.5}{\milli\Omega}$ and forward voltage of $U_{\mathrm{D}} = \SI{1}{\volt}$ for the diode and an equivalent resistance of $R_{\mathrm{T}} = \SI{30}{\milli\Omega}$ for the transistor. Determine the required duty cycle value when these additional semiconductor conduction losses are considered.}

\begin{solutionblock}
	Considering the additional losses, the power balance equation is extended accordingly
	\begin{equation}
		U_1 I_{\mathrm{L}} = U_2 I_2 + R_{\mathrm{L}} I_{\mathrm{L}}^2 + R_{\mathrm{T}} I_{\mathrm{T}}^2 + U_{\mathrm{D}} \overline{i}_{\mathrm{D}} + R_{\mathrm{D}} I_{\mathrm{D}}^2,
		\label{eq:powerBalanceBoostConverterWithLosses}
	\end{equation}
	with the average diode current $\overline{i}_{\mathrm{D}}$ and the RMS diode current $I_{\mathrm{D}}$. The diode RMS current is determined as:
	\begin{equation}
		I_{\mathrm{D}} = \sqrt{\frac{1}{T_\mathrm{s}} \int_{0}^{T_\mathrm{s}} i_{\mathrm{D}}(t)^2 \mathrm{d}t}
		= \sqrt{\frac{1}{T_\mathrm{s}} \left( \int_{0}^{T_{\mathrm{on}}} 0 \mathrm{d}t + \int_{T_{\mathrm{on}}}^{T_{\mathrm{s}}} i_{\mathrm{L}}(t)^2 \mathrm{d}t \right)}
		= \sqrt{\frac{{T_{\mathrm{s}}-T_{\mathrm{on}}}}{T_{\mathrm{s}}}I_{\mathrm{L}}^2}
		= \sqrt{1-D}I_{\mathrm{L}}.
	\end{equation}
	Moreover, the average diode current is calculated with
	\begin{equation}
		\overline{i}_{\mathrm{D}} = (1-D)I_{\mathrm{L}} = I_2,
		\label{eq:avgCurrentDiode}
	\end{equation}
	and, the RMS current through the transistor is given by:
	\begin{equation}
		I_{\mathrm{T}} = \sqrt{\frac{1}{T_\mathrm{s}} \int_{0}^{T_\mathrm{s}} i_{\mathrm{T}}(t)^2 \mathrm{d}t}
		= \sqrt{\frac{1}{T_{\mathrm{s}}}\left(\int_{0}^{T_{\mathrm{on}}} I_{\mathrm{L}}^2 \mathrm{d} t + \int_{T_{\mathrm{on}}}^{T_{\mathrm{s}}} 0 \mathrm{d} t \right)}
		= \sqrt{\frac{T_{\mathrm{on}}}{T_{\mathrm{s}}}I_{\mathrm{L}}^2}
		= \sqrt{D} I_{\mathrm{L}}.
	\end{equation}
	Next, the currents in \eqref{eq:powerBalanceBoostConverterWithLosses} are replaced with $I_{\mathrm{2}}$, which leads to
	\begin{equation}
		\frac{U_1 I_2}{1 -D} = U_2 I_2 + R_{\mathrm{L}}\frac{I_2^2}{\left(1-D\right)^2} + R_{\mathrm{T}} \frac{D I_2^2}{\left(1-D\right)^2} + U_{\mathrm{D}}I_2 + R_{\mathrm{D}}\frac{I_2^2}{1-D},
	\end{equation}
	with $I_2 = \frac{U_2}{R}$ and $D_{\mathrm{p}} = 1-D$ yields:
	\begin{equation}
		U_1 D_{\mathrm{p}} = U_2 D_{\mathrm{p}}^2 + \frac{R_{\mathrm{L}}}{R} U_2 + (1-D_{\mathrm{p}}) \frac{R_{\mathrm{T}}}{R}U_2 + U_{\mathrm{D}}D_{\mathrm{p}}^2 + \frac{R_{\mathrm{D}}}{R}U_2 D_{\mathrm{p}}.
	\end{equation}
	Rearrange the equation into the quadratic formula structure by
	\begin{equation}
		0 = \left(U_2 + U_{\mathrm{D}} \right)D_{\mathrm{p}}^2 + \left(\frac{R_{\mathrm{D}}}{R}U_2 - \frac{R_{\mathrm{T}}}{R}U_2 -U_1\right)D_{\mathrm{p}} + \frac{R_{\mathrm{T}}+R_{\mathrm{L}}}{R} U_2,
	\end{equation}
	leads to the first solution given by
	\begin{equation}
		D_{\mathrm{P3}} = \frac{1}{2} \frac{U_1 +\frac{R_{\mathrm{T}}}{R}U_2 - \frac{R_{\mathrm{D}}}{R}U_2}{U_2 + U_{\mathrm{D}}} + \sqrt{\left(\frac{1}{2} \frac{\frac{R_{\mathrm{T}}}{R}U_2 - \frac{R_{\mathrm{D}}}{R}U_2 + U_1}{U_2 + U_{\mathrm{D}}}\right)^2 - \frac{\frac{R_{\mathrm{T}}+R_{\mathrm{L}}}{R}U_2}{U_2+U_{\mathrm{D}}}}
		= 0.146,
	\end{equation}
	which results in a duty cycle of $D_3 = 0.854$.	The second solution is calculated in the same way as follows
	\begin{equation}
		D_{\mathrm{P4}} = \frac{1}{2} \frac{U_1 +\frac{R_{\mathrm{T}}}{R}U_2 - \frac{R_{\mathrm{D}}}{R}U_2}{U_2 + U_{\mathrm{D}}} - \sqrt{\left(\frac{1}{2} \frac{\frac{R_{\mathrm{T}}}{R}U_2 - \frac{R_{\mathrm{D}}}{R}U_2 + U_1}{U_2 + U_{\mathrm{D}}}\right)^2 - \frac{\frac{R_{\mathrm{T}}+R_{\mathrm{L}}}{R}U_2}{U_2+U_{\mathrm{D}}}}
		= 0.0516,
	\end{equation}
	resulting in $D_4 = 0.948$.
\end{solutionblock}

\subtask{Beside the conduction losses, also switching losses need to be considered in practice. In \autoref{fig:TurnOffSiliconDiode} the voltage and current waveforms are visualized for the turn-off event of a diode (reverse recovery effect). Therefore, calculate the turn-off losses for a fast diode with a commutation loop inductance of $L_{\mathrm{c}} = \SI{500}{\nano\henry}$. The peak reverse recovery current is $\hat{i}_{\mathrm{rr}} = \SI{4}{\ampere}$ and the reverse recovery time is $T_{\mathrm{rr}} = t_2 - t_0 = \SI{46.6}{\nano\second}$.}

\input{fig/ex02/Fig_diode_switchOff.tex}

\begin{solutionblock}
	Diode switching losses occur when the diode voltage and current are not zero during commutation. In \autoref{fig:TurnOffSiliconDiode} the voltage and current waveforms are shown. The voltage $u_{\mathrm{D}}$ is for the interval $t_0$ until $t_1$ small and, therefore, the switching losses. At the time step $t_1$ the diode is blocking and the peak reverse voltage $U_{\mathrm{RB}}$ applies. Due to this much bigger value, the switching losses increases significantly, resulting that only the interval between $t_1$ and $t_2$ is responsible for the switching losses. Hence, the losses are calculated as follows
	\begin{equation}
		P_{\mathrm{l}} = \frac{1}{T} \int_{t_1}^{t_2} i_{\mathrm{rr}} U_{\mathrm{RB}} \mathrm{d}t = \frac{1}{T} U_{\mathrm{RB}} \int_{t_1}^{t_2} i_{\mathrm{rr}} \mathrm{d}t,
	\end{equation}
	with the electric charge
	\begin{equation}
		Q_2 = \int_{t_1}^{t_2} i_{\mathrm{rr}} \mathrm{d}t,
		\label{eq:chargeQ2}
	\end{equation}
	which results to the switching loss (per period) as:
	\begin{equation}
		P_{\mathrm{l}} = \frac{U_{\mathrm{RB}}Q_2}{T_\mathrm{s}}.
	\end{equation}

	With the assumption of a linear current waveform, the current slopes are defined as
	\begin{equation}
		\beta = \left|\frac{\mathrm{d}i_{\mathrm{D}}(t)}{\mathrm{d}t}\right|, \quad t_0 < t \leq t_1 \qquad \gamma = \frac{\mathrm{d}i_{\mathrm{D}}(t)}{\mathrm{d}t}, \quad t_1 < t \leq t_2.
	\end{equation}

	The equivalent circuit diagram of the diode switch-off event is shown in \autoref{fig:DiodeSwitchOff}.
	\input{fig/ex02/sFig_diode_sw_off_esb.tex}
	The current $i_{\mathrm{D}}(t)$ is dependent on the voltage $U_2$ by:
	\begin{equation}
		\beta = \frac{U_2}{L_{\mathrm{c}}} = \frac{\SI{60}{\volt}}{\SI{500}{\nano\henry}} = \SI{120}{\ampere\per\micro\second}.
	\end{equation}

	With the current slope $\beta$ during the time interval $T_{10} = t_1 - t_0$, the maximum reverse current $\hat{i}_{\mathrm{rr}}$ results in
	\begin{equation}
		\beta T_{10} = \hat{i}_{\mathrm{rr}},
	\end{equation}
	which is rearranged to calculate the time interval with:
	\begin{equation}
		T_{10} = \frac{\hat{i}_{\mathrm{rr}}}{\beta} = \frac{\SI{4}{\ampere}}{\SI{120}{\ampere\per\micro\second}} = \SI{33.3}{\nano\second}.
	\end{equation}

	Hence, the time interval $T_{21}$ is given with:
	\begin{equation}
		T_{21} = T_{\mathrm{rr}} - T_{10} = \SI{46.6}{\nano\second} - \SI{33.3}{\nano\second} = \SI{13.3}{\nano\second}.
	\end{equation}

	With \eqref{eq:chargeQ2} the electrical charge $Q_2$ is calculated by:
	\begin{equation}
		Q_2 = \frac{1}{2} T_{21} \hat{i}_{\mathrm{rr}} = \frac{\SI{13.3}{\nano\second}}{2} \cdot \SI{4}{\ampere} = \SI{26.6}{\nano\coulomb}.
	\end{equation}

	Reverse breakdown voltage is determined form the ECD (\autoref{fig:DiodeSwitchOff}) shortly after $t_1$ with the second Kirchhoff's law:
	\begin{equation}
		U_{\mathrm{RB}} = U_2 + L_{\mathrm{c}} \frac{\mathrm{d}i_{\mathrm{rr}}(t)}{\mathrm{d}t} = U_2 + L_{\mathrm{c}} \gamma = \SI{60}{\volt} + \SI{500}{\nano\henry} \frac{\SI{4}{\ampere}}{\SI{13.3}{\nano\second}} = \SI{210.4}{\volt}.
		\label{eq:reverseBreakdownVoltage}
	\end{equation}

	Therefore, the turn-off switching loss (average per period) is resulting in:
	\begin{equation}
		P_{\mathrm{l,D1}} = \frac{U_{\mathrm{RB}}Q_2}{T_{\mathrm{s}}} = \frac{\SI{210.4}{\volt}\cdot \SI{26.6}{\nano\coulomb}}{\frac{1}{\SI{100}{\kilo\hertz}}} = \SI{0.56}{\watt}.
	\end{equation}

\end{solutionblock}

\task{Buck-boost converter}
A wide input-to-output voltage range can be realized by the cascade of buck and boost converters with a common inductance.
With this topology the output voltage can be adjusted to a value which is higher or lower than the input voltage.
The output voltage is kept at the specified constant value by adjusting the duty cycles $D_1$
(of transistor  $T_1$) and $D_2$ (of transistor $T_2$) using a control system. Both transistors operate at the same switching frequency.
The ripple of the output voltage and of the current in the inductor can be ignored. The current in $L$ is continuous.
Initially, both transistors operate with the same duty cycle $D_1 = D_2 = D$ and their switching patterns are synchronized.

\subtask{Calculate the duty cycle of the transistors $T_1$ and $T_2$ depending on the voltage transformation ratio $U_\mathrm{2}/U_\mathrm{1}$.}

\begin{solutionblock}
	% Solution
	From the lecture script the voltage transformation ratio of a buck-boost converter is given by
	\begin{equation}
		U_\mathrm{2}=\frac{D}{1-D} U_\mathrm{1}
		\hspace{1cm}.
	\end{equation}
	Hence, the duty cycle $D$ for a specific voltage transformation ratio can be calculated as
	\begin{equation}
		D = \frac{\frac{U_\mathrm{2}}{U_\mathrm{1}}} {1+{\frac{U_\mathrm{2}}{U_\mathrm{1}}}}.
		\label{eq:DutyCycle}
	\end{equation}
\end{solutionblock}

\subtask{How should $D_1$ and $D_2$ be selected so that the losses of the overall system are minimal?
	The relationships calculated under subtask 2.3.4 and 2.3.5 can be used for the voltage transformation ratio and the value of $\overline{i}_\mathrm{L}$.}
\begin{solutionblock}
	% Solution
	The losses of the transistors are:
	\begin{equation}
		P_\mathrm{loss,T1}=D_1 U_\mathrm{F} \overline{i}_\mathrm{L}=D_1 U_\mathrm{F} \frac{P_\mathrm{2}}{U_\mathrm{2}} \cdot \frac{1}{1-D_2}
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		P_\mathrm{loss,T2}=D_2 U_\mathrm{F} \overline{i}_\mathrm{L}=D_2 U_\mathrm{F} \frac{P_\mathrm{2}}{U_\mathrm{2}} \cdot \frac{1}{1-D_2}.
	\end{equation}
	So the losses of both transitors can be expressed by
	\begin{equation}
		P_\mathrm{loss}=\left(D_1 +D_2\right) U_\mathrm{F} \frac{P_\mathrm{2}}{U_\mathrm{2}} \cdot \frac{1}{1-D_2}
	\end{equation}
	and with
	\begin{equation}
		D_1 =\frac{U_\mathrm{2}}{U_\mathrm{1}} \left(1 - D_2 \right)
		\hspace{1cm} \Rightarrow \hspace{1cm}
		P_\mathrm{loss}=\left(\frac{U_\mathrm{2}}{U_\mathrm{1}} \left(1 - D_2 \right)+D_2\right) U_\mathrm{F} \frac{P_\mathrm{2}}{U_\mathrm{2}} \cdot \frac{1}{1-D_2}.
	\end{equation}
	This leads to
	\begin{equation}
		P_\mathrm{loss}=P_\mathrm{2} U_\mathrm{F} \left(\frac{1}{U_\mathrm{1}} +
		\frac{1}{U_\mathrm{2}} \cdot \frac{D_2}{1-D_2}\right).
		\label{eq:ploss}
	\end{equation}
	The losses minimum is determined by differentiating with respect to the duty cycle $D_2$.
	\begin{equation}
		\frac{\mathrm{d}}{\mathrm{d}D_2} P_\mathrm{loss}(D_2)= \frac{P_2 U_\mathrm{F}}{U_2}\frac{1}{\left(1 - D_2 \right)^2} > 0.
		\label{eq:derivateDutycyle}
	\end{equation}
	For any feasible $D_2$ the exact root of \eqref{eq:derivateDutycyle} does not exist, which means that the power loss is a strictly increasing function of $D_2$. Hence, to reduce losses we can conclude the following:
	\begin{itemize}
		\item Activate transistor $T_2$ only if boost mode is required (otherwise $D_2=0$ as default).
		\item Set $D_2$ as small as possible if operation requires boost mode, i.e., $D_1$ should be as large as possible (in this case).
	\end{itemize}
\end{solutionblock}

\subtask{How high is the maximum efficiency gain and at which operating point does it occur? Give an explanation
	for the observed finding.}

\begin{solutionblock}
	% Solution
	The efficiency gain is calculated by $\Delta \eta$=$\eta_\mathrm{opt}$-$\eta_\mathrm{sync}$.
	You get following result:

	\input{./fig/ex02/sFigTab_EfficiencyGain}

	The maximum efficiency gain is at $U_\mathrm{1}=\SI{400}{\volt}$.
	It looks like the efficiency gain is higher within boost mode, but why is the highest efficiency gain at $\SI{400}{\volt}$?
	Let's consider the power loss at boost mode for the two cases. The power loss needs to be expressed by a function of the voltages:

	\begin{equation}
		P_\mathrm{loss}=D_1 U_\mathrm{F} \overline{i}_\mathrm{L} + D_2 U_\mathrm{F} \overline{i}_\mathrm{L}.
		\label{eq:ploss2}
	\end{equation}
	Considering the optimal efficiency the $D_1=1$ and $D_2=f(U_\mathrm{1})$ for $\SI{320}{\volt} \leq U_\mathrm{1} \leq \SI{400}{\volt}$
	(boost mode) and $D_2=0$ and $D_1=f(U_\mathrm{1})$ for $\SI{400}{\volt} \le U_\mathrm{2} \leq \SI{720}{\volt}$ (buck mode). \\
	For $\SI{320}{\volt} \leq U_\mathrm{1} \leq \SI{400}{\volt}$ (boost mode)
	\begin{equation}
		\overline{i}_\mathrm{L}=\frac{P_\mathrm{2}}{U_\mathrm{1}}
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		D_1=1
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		D_2=1-\frac{U_\mathrm{1}}{U_\mathrm{2}}
	\end{equation}
	leads to
	\begin{equation}
		P_\mathrm{loss,opt}=\frac{2 P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{1}} - \frac{P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{2}}.
		\label{eq:Plossoptboost}
	\end{equation}
	For $\SI{400}{\volt} \le U_\mathrm{1} \leq \SI{720}{\volt}$ (buck mode)
	\begin{equation}
		\overline{i}_\mathrm{L}=\frac{P_\mathrm{2}}{U_\mathrm{2}}
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		D_1=\frac{U_\mathrm{2}}{U_\mathrm{1}}
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		D_2=0
	\end{equation}
	leads to
	\begin{equation}
		P_\mathrm{loss,opt}=\frac{P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{1}}.
		\label{eq:Plossoptbuck}
	\end{equation}

	For the synchronous operation mode following is valid:
	\begin{equation}
		\overline{i}_\mathrm{L}=\frac{P_\mathrm{2}}{U_\mathrm{2}} \cdot \frac{1}{1-D}
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		D_1=D_2=D
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		D=\frac{U_\mathrm{2}}{U_\mathrm{1}+U_\mathrm{2}}.
	\end{equation}
	Using \eqref{eq:ploss2} leads to:
	\begin{equation}
		P_\mathrm{loss,sync}=\frac{P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{1}} + \frac{P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{1}}=\frac{2 P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{1}}.
		\label{eq:Plosssync}
	\end{equation}

	The efficiency can be expressed by
	\begin{equation}
		\eta = \frac{P_\mathrm{2}}{P_\mathrm{loss}+P_\mathrm{2}}=\frac{1}{1+\frac{P_\mathrm{loss}}{P_\mathrm{2}}}.
		\label{eq:efficiencysyncopt}
	\end{equation}
	Using \eqref{eq:Plossoptboost} and \eqref{eq:Plossoptbuck} in \eqref{eq:efficiencysyncopt} results in:
	\begin{equation}
		\eta_\mathrm{opt,boost} = \frac{1}{1+U_\mathrm{F} \left(\frac{2}{U_\mathrm{1}}-\frac{1}{U_\mathrm{2}}\right)}
		\hspace{1cm} \mathrm{and} \hspace{1cm}
		\eta_\mathrm{opt,buck} = \frac{1}{1+\frac{U_\mathrm{F}}{U_\mathrm{1}}}.
		\label{eq:efficiencyopt}
	\end{equation}
	Using \eqref{eq:Plosssync} in \eqref{eq:efficiencysyncopt} results in:
	\begin{equation}
		\eta_\mathrm{sync} = \frac{1}{1+U_\mathrm{F} \frac{2}{U_\mathrm{1}}}.
		\label{eq:efficiencysync}
	\end{equation}

	The efficiency gain is $\Delta \eta=\eta_\mathrm{opt}-\eta_\mathrm{sync}$:

	\begin{equation}
		\Delta \eta_\mathrm{boost}= \frac{\frac{U_\mathrm{F}}{U_\mathrm{2}}}{\left(1+ \frac{2 U_\mathrm{F}}{U_\mathrm{1}}-\frac{U_\mathrm{F}}{U_\mathrm{2}} \right) \left(1+\frac{2 U_\mathrm{F}}{U_\mathrm{1}}\right)}.
		\label{eq:efficiencygainboost}
	\end{equation}

	\begin{equation}
		\Delta \eta_\mathrm{buck}= \frac{\frac{U_\mathrm{F}}{U_\mathrm{1}}}{\left(1+ \frac{2 U_\mathrm{F}}{U_\mathrm{1}}\right) \left(1+\frac{ U_\mathrm{F}}{U_\mathrm{1}}\right)}.
		\label{eq:efficiencygainbuck}
	\end{equation}

	The two efficiencies are calculated with \eqref{eq:efficiencyopt},\eqref{eq:efficiencysync}  and displayed in \autoref{fig:EfficiencyAtInputVoltage}.

	Conclusions:
	\begin{itemize}
		\item \eqref{eq:efficiencysync} and \eqref{eq:efficiencyopt} shows, that $\eta_\mathrm{opt,boost}$
		      and $\eta_\mathrm{sync}$ differ not simply by a constant factor.
		\item \eqref{eq:efficiencysync} and \eqref{eq:efficiencyopt} shows, that $\eta_\mathrm{opt,buck}$
		      and $\eta_\mathrm{sync}$ differ in the coefficient of $\frac{U_\mathrm{F}}{U_\mathrm{1}}$ in the denominator.
		      This cause that the difference decrease with increasing $U_\mathrm{1}$.
		\item The difference between the power losses of $P_\mathrm{loss,opt}$ and $P_\mathrm{loss,sync}$ in boost mode is
		      $\frac{P_\mathrm{2} U_\mathrm{F}}{U_\mathrm{2}}$ (constant)
		      which has a proportionally greater effect on the efficiency with smaller total power losses at larger $U_\mathrm{1}$.
		\item The highest efficiency gain is obtained at $U_\mathrm{1}=\SI{400}{\volt}$. In boost operation, the absolute conduction-loss reduction is constant, while the efficiency gain increases with increasing input voltage. In buck operation, the efficiency gain decreases with increasing input voltage. Therefore, the maximum occurs at the transition between boost and buck operation under the considered loss model.
	\end{itemize}
